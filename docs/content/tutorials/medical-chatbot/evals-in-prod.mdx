In this section we'll learn how to set up tracing for our medical chatbot to observe it on a component level and ensure your chatbot performs well and gets full visibility for debugging internal components.

In the development section of this tutorial, we've already added `@observe` decorator to our chatbot's components, now we will add metrics and spans to this tracing setup to enable evaluations.

## Setup Tracing

`deepeval` offers an `@observe` decorator for you to apply metrics at any point in your LLM app to evaluate any [LLM interaction](https://deepeval.com/docs/evaluation-test-cases#what-is-an-llm-interaction),
this provides full visibility for debugging internal components of your LLM application. [Learn more about tracing here](https://deepeval.com/docs/evaluation-llm-tracing).

To add metrics and spans to your traces, modify your `MedicalChatbot` class like this:

import { QdrantClient } from "@qdrant/js-client-rest";
import { ChatOpenAI, OpenAIEmbeddings } from "@langchain/openai";
import { observe, updateCurrentSpan, updateCurrentTrace } from "deepeval/tracing";
import { metrics } from "deepeval";
import * as readline from "node:readline/promises";

class MedicalChatbot {
  model: ChatOpenAI;
  appointments: Record<string, any>;
  encoder: OpenAIEmbeddings;
  client: QdrantClient;
  systemPrompt: string;

  constructor({
    documentPath,
    model = "gpt-4",
    encoder = "text-embedding-3-small",
    memory = "http://localhost:6333",
    systemPrompt = "",
  }: {
    documentPath: string;
    model?: string;
    encoder?: string;
    memory?: string;
    systemPrompt?: string;
  }) {
    this.model = new ChatOpenAI({ model });
    this.appointments = {};
    this.encoder = new OpenAIEmbeddings({ model: encoder });
    this.client = new QdrantClient({ url: memory });
    this.storeData(documentPath);
    this.systemPrompt =
      systemPrompt ||
      "You are a virtual health assistant designed to support users with symptom understanding and appointment management. Start every conversation by actively listening to the user's concerns. Ask clear follow-up questions to gather information like symptom duration, intensity, and relevant health history. Use available tools to fetch diagnostic information or manage medical appointments. Never assume a diagnosis unless there's enough detail, and always recommend professional medical consultation when appropriate.";
    this.setupAgent(this.systemPrompt);
  }

  storeData(documentPath: string) {
    // ...
  }

  // A tool to retrieve data on various diagnosis methods from gale encyclopedia
  queryEngine = observe({
    type: "retriever",
    metrics: [new metrics.ContextualRelevancyMetric()],
    fn: async (query: string) => {
      const { points: hits } = await this.client.query("gale_encyclopedia", {
        query: await this.encoder.embedQuery(query),
        limit: 3,
        with_payload: true,
      });

      const contexts = hits.map((hit) => hit.payload!.content as string);

      // Here, updateCurrentSpan() will update the Retriever span
      updateCurrentSpan({
        input: query,
        retrievalContext: contexts,
      });
      return contexts.join("\n");
    },
  });

  // ... Other tools here

  interactiveSession = observe({
    type: "agent",
    fn: async (sessionId: string) => {
      console.log("Hello! I am Baymax, your personal health care companian.");
      console.log("Please enter your symptoms or ask about appointment details. Type 'exit' to quit.");

      const rl = readline.createInterface({ input: process.stdin, output: process.stdout });

      while (true) {
        const userInput = await rl.question("Your query: ");
        if (userInput.toLowerCase() === "exit") {
          break;
        }

        const response = await this.agentWithChatHistory.invoke(
          { input: userInput },
          { configurable: { sessionId } },
        );
        updateCurrentTrace({
          threadId: sessionId,
          input: userInput,
          output: response.output,
        });
        console.log("Agent Response:", response.output);
      }

      rl.close();
    },
  });
}
